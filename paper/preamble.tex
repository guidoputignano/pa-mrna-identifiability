%  Shared grammar for the three model block diagrams, Sections 3, 4 and 5.
%  Defined once so the three are directly comparable: the pool m, the load

% A miniature of a channel's shape, drawn beside its box. `lin' for a
% straight line, `sat' for a saturating curve.
% #3 = l linear | s saturating | n rising then falling. Model C's alternative
% channel is the third shape, and it is the whole difference between C and B.
\newcommand{\channelglyph}[3]{%
  \begin{scope}[shift={#1}, scale=0.62]
    \draw[gray!55, line width=0.3pt] (0,0) -- (1.15,0) (0,0) -- (0,0.8);
    \if#3l \draw[#2, line width=1.1pt] (0,0) -- (1.05,0.72);
    \else\if#3n \draw[#2, line width=1.1pt]
              plot[smooth] coordinates {(0,0) (0.24,0.46) (0.52,0.64) (0.80,0.48) (1.05,0.20)};
    \else   \draw[#2, line width=1.1pt]
              plot[smooth] coordinates {(0,0) (0.26,0.36) (0.58,0.58) (1.05,0.70)};
    \fi\fi
  \end{scope}}
